    pmon_core
    PUBLIC
        IS_DEBUG=$<IF:$<CONFIG:Debug>,1,0>
    PRIVATE
        _CRT_SECURE_NO_WARNINGS
)

    pmon_core
    PUBLIC
        "${PROJECT_SOURCE_DIR}"

        # <PresentMonAPI2/PresentMonAPI.h>.
        "${PROJECT_SOURCE_DIR}/IntelPresentMon"
)

    PUBLIC

)

    pmon_core

        /fp:fast
        $<$<CONFIG:Release>:/guard:cf>
)

    pmon_core
    INTERFACE
        $<$<CONFIG:Release>:/guard:cf>
)

    pmon_core
    PROPERTIES
        OUTPUT_NAME Core
        FOLDER "UI and Capture"
)
